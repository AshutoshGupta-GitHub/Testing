-- creating the before trigger
Create trigger BeforeInsertTrigger
on emps
instead of insert
as
begin
	declare @sal decimal(9,2);
	select @sal = salary from inserted;
	if(@sal < 10000)
	begin
		print 'Salary should be more than 10000'
	end
	else
	begin
		insert into emps(name, salary) values(null, @sal);
	end
end

--Cheking if name is present then don't insert if is it not then insert,
Create OR ALTER trigger BeforeTriggerInsert
on emps2
instead of insert
as
begin
	declare @newname varchar(100);
	select @newname = name from emps2 where name=@newname;
  
	declare @cnt int;
	--select @name = name from inserted
	--where @name = name;
	select @cnt=count(name) from inserted Group by name;
	
	if(@cnt > 1)
	begin
		print 'Name is already present Duplicate Value!! insert Unique one';
	end
	else
	begin
		insert into emps2(name, salary) select name,salary from inserted;
	end
end
