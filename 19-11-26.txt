--FInding the netsal using Scalar f^ and Multi stat. F^ and SP
--Creating scalar F^

create OR ALTER function findPF(@salary decimal(9,2))
returns decimal(9,2)
as 
begin 
	declare @pfamt decimal (9,2);
	set @pfamt = @salary * 0.1;
	return @pfamt;
end

--Multi Statement F^
Create OR ALTER function findEmp(@Name varchar(100), @salary decimal(9,2))
returns @temptb table
(names varchar(100), salary decimal(9,2), pfamt decimal(9,2), netsal decimal(9,2))
as
begin
	insert into @temptb (names, salary,pfamt) values (@Name, @salary, dbo.findPF(@salary));
	declare @netsal decimal(9,2);
	set @netsal = @salary - dbo.findPF(@salary);
	update @temptb set netsal = @netsal;
	return
end

--Creating SP
Create OR ALTER proc SP_saveEmp
@Salary decimal(9,2),
@Name varchar(100)
as
begin
	DECLARE @nextEid INT;
    SELECT @nextEid = ISNULL(MAX(eid), 0) + 1 FROM emp;

	insert into emp(ename,esalary, pf,netsal) 
	select @nextEid, names, salary,pfamt,netsal 
	dfrom dbo.findEmp(@Name,@salary);
end
